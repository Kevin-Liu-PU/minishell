cmake_minimum_required(VERSION 3.16)
project(minishell VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(fsh_parser src/parser.cpp)
target_include_directories(fsh_parser PUBLIC include)
add_executable(parser_tests tests/parser_tests.cpp)
target_link_libraries(parser_tests PRIVATE fsh_parser)
if(MSVC)
    target_compile_options(fsh_parser PRIVATE /W4)
    target_compile_options(parser_tests PRIVATE /W4)
else()
    target_compile_options(fsh_parser PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(parser_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_test(NAME parser COMMAND parser_tests)
if(UNIX)
    add_executable(fsh src/main.cpp)
    target_link_libraries(fsh PRIVATE fsh_parser)
    target_compile_options(fsh PRIVATE -Wall -Wextra -Wpedantic)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:fsh>)
        set_tests_properties(integration PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
    endif()
else()
    message(STATUS "POSIX executor disabled: native Windows builds only the parser and its tests")
endif()
